(defun wilczek/org-buffer-stats ()
  "Calculates cryptographic hash and word count for the current buffer."
  (interactive)
  (let* ((text (buffer-string))
         (words (count-words (point-min) (point-max)))
         (hash (secure-hash 'sha256 text)))
    (message "Buffer: %s | Words: %d | SHA256: %s" (buffer-name) words hash)))
#+END_SRC
